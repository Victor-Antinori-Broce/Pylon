### API del Query Engine
```typescript
// Query básica
const result = await parseQuery({
  table: agentIdentities,
  where: { isActive: true },
  limit: 10,
});

// Query con expansiones (Anti N+1)
const result = await parseQuery({
  table: agentIdentities,
  where: { type: "service" },
  expand: ["traces", "apiKeys", "memories"],
  orderBy: [{ field: "createdAt", direction: "desc" }],
  limit: 10,
  offset: 0,
});
// Retorna: { data: [{ id, name, traces: [...], apiKeys: [...] }], meta: {...} }

// Validación previa
const validation = validateQuery({
  table: agentIdentities,
  where: { status: "active" },
  expand: ["traces"],
});
// Retorna: { valid: true, errors: [] }
```
